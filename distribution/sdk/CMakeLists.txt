cmake_minimum_required(VERSION 3.20)
project(MermanSdkConsumer LANGUAGES C)
option(MERMAN_BUILD_CPP_EXAMPLES "Build C++17 shared and static consumers" OFF)

find_package(Merman CONFIG REQUIRED)
foreach(consumer IN ITEMS render probe)
  add_executable(merman-${consumer} examples/${consumer}.c)
  target_compile_features(merman-${consumer} PRIVATE c_std_99)
  target_link_libraries(merman-${consumer} PRIVATE Merman::Merman)
endforeach()
add_executable(merman-render-static examples/render.c)
target_compile_features(merman-render-static PRIVATE c_std_99)
target_link_libraries(merman-render-static PRIVATE Merman::Static)
enable_testing()
add_test(NAME sdk-render COMMAND merman-render)
add_test(NAME sdk-render-static COMMAND merman-render-static)
if(MERMAN_BUILD_CPP_EXAMPLES)
  enable_language(CXX)
  foreach(linkage IN ITEMS shared static)
    add_executable(merman-render-cpp-${linkage} examples/render.cpp)
    target_compile_features(merman-render-cpp-${linkage} PRIVATE cxx_std_17)
  endforeach()
  target_link_libraries(merman-render-cpp-shared PRIVATE Merman::Merman)
  target_link_libraries(merman-render-cpp-static PRIVATE Merman::Static)
  add_test(NAME sdk-cpp-shared COMMAND merman-render-cpp-shared "${CMAKE_CURRENT_BINARY_DIR}/cpp-shared")
  add_test(NAME sdk-cpp-static COMMAND merman-render-cpp-static "${CMAKE_CURRENT_BINARY_DIR}/cpp-static")
endif()
